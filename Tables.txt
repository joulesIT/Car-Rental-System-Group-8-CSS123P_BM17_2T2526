Table: bookings
Columns:
id int(11) AI PK 
username varchar(100) 
car_id int(11) 
start_date date 
end_date date 
rental_days int(11) 
total_amount double 
booked_at timestamp 
total_price double 
payment_method varchar(20)


Table: cars
Columns:
id int(11) AI PK 
brand varchar(50) 
model varchar(50) 
rating double 
image_path varchar(255) 
available tinyint(1) 
description text 
type varchar(50) 
price_per_day decimal(10,2) 
seats int(11) 
fuel_type varchar(50) 
is_deleted tinyint(1)

Table: rentals
Columns:
id int(11) AI PK 
car_id int(11) 
username varchar(50) 
start_date date 
end_date date 
total_price decimal(10,2) 
payment_method varchar(20)

Table: user_cards
Columns:
id int(11) AI PK 
username varchar(50) 
card_number varchar(16) 
expiry_date varchar(5) 
cvv varchar(3) 
balance double

Table: users
Columns:
ID int(11) AI PK 
USERNAME varchar(50) 
EMAIL varchar(200) 
password varchar(50) 
legal_name varchar(100) 
drivers_license varchar(50) 
phone_number varchar(20) 
role varchar(10) 
EMPLOYEE_ID varchar(50)
